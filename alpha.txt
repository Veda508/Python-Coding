Question was : Given an array arr[] of integers and an integer k, your task is to find the maximum value for each contiguous subarray of size k. The output should be an array of maximum values corresponding to each contiguous subarray.

answer:
def maxOfSubarrays(arr, k):
    n = len(arr)
    res = []
    for i in range(0, n - k + 1):
        max = arr[i]
        for j in range(1, k):
            if arr[i + j] > max:
                max = arr[i + j]
        res.append(max)
    return res

if __name__ == "__main__":
    arr = [ 1, 2, 3, 1, 4, 5, 2, 3, 6 ]
    k = 3
    res = maxOfSubarrays(arr, k)
    for maxVal in res:
        print(maxVal, end=" ")
